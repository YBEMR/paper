def sha256(data):
    return hashlib.sha256(data).hexdigest()


def _replace(data, old, new):
    old, new = old.encode("utf-8"), new.encode("utf-8")
    if data.count(old) != 1:
        raise ValueError("Expected exactly one immutable byte anchor")
    return data.replace(old, new, 1)


def instrument_parent(data):
    if sha256(data) != PARENT_SHA256:
        raise ValueError("Parent SHA256 mismatch; no patch applied")
    for old, new in REPLACEMENTS:
        data = _replace(data, old, new)
    return data


# Public deterministic API for the parent freeze manifest.
transform = instrument_parent

def restore_parent(data):
    for old, new in reversed(REPLACEMENTS):
        data = _replace(data, new, old)
    if sha256(data) != PARENT_SHA256:
        raise ValueError("Reversed source is not the immutable parent")
    return data


TRIGGERS = ("request_deadline", "worker_guard", "cancel_file", "stop_file")


def simulate_original(probes, before_decision=None):
    """Python reference for original callback catch+OR, not MATLAB execution."""
    try:
        if before_decision is not None:
            before_decision()
        return bool(probes[0]() or probes[1]() or probes[2]() or probes[3]())
    except Exception:
        return True


def simulate_instrumented(probes, *, state="iter", event_number=1,
                          before_decision=None, logger=None):
    """Software surrogate tests probe order, short circuit and logger isolation."""
    triggers = {name: {"attempted": False, "evaluated": False, "value": None}
                for name in TRIGGERS}
    stop, original_error = False, None
    try:
        if before_decision is not None:
            before_decision()
        for index, name in enumerate(TRIGGERS):
            triggers[name]["attempted"] = True
            value = bool(probes[index]())
            triggers[name].update(evaluated=True, value=value)
            if value:
                stop = True
                break
    except Exception as error:
        stop, original_error = True, type(error).__name__
    for record in triggers.values():
        record["not_evaluated_reason"] = ("" if record["evaluated"] else
            "original_callback_exception" if original_error else "short_circuit")
    record = {"actual_stop": stop, "state": state, "event_number": event_number,
              "triggers": triggers, "original_callback_exception": original_error is not None}
    result = {"actual_stop": stop, "record": deepcopy(record), "log_complete": True, "errors": []}
    try:
        if logger is not None:
            logger(deepcopy(record))
    except Exception as error:
        result["log_complete"] = False
        result["errors"].append(type(error).__name__)
    return result


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--source", type=Path)
    parser.add_argument("--check", action="store_true")
    args = parser.parse_args(argv)
    if args.check:
        data = TARGET.read_bytes()
        if sha256(data) != TARGET_SHA256:
            raise ValueError("Instrumented worker differs from frozen target")
        restore_parent(data)
        print(json.dumps({"status": "CHECK_PASS", "sha256": sha256(data)}))
        return
    if args.source is None:
        raise ValueError("Explicit original --source required; never repatch an unknown worker")
    parent = args.source.read_bytes()
    patched = instrument_parent(parent)
    if TARGET_SHA256 != "TO_BE_FROZEN" and sha256(patched) != TARGET_SHA256:
        raise ValueError("Generated target SHA256 mismatch")
    existing = TARGET.read_bytes()
    if existing not in (parent, patched):
        raise ValueError("Destination differs from parent and expected target; refusing overwrite")
    if existing != patched:
        TARGET.write_bytes(patched)
    print(json.dumps({"status": "GENERATED", "parent_sha256": sha256(parent),
                      "sha256": sha256(patched), "target": str(TARGET.relative_to(ROOT))}))


if __name__ == "__main__":
    main()
