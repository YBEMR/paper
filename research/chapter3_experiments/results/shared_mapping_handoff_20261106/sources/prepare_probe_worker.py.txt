def sha256(raw):
    return hashlib.sha256(raw).hexdigest()


def _replace(raw, old, new):
    old, new = old.encode("utf-8"), new.encode("utf-8")
    if raw.count(old) != 1:
        raise ValueError("Expected one immutable byte anchor")
    return raw.replace(old, new, 1)


def transform(raw):
    if sha256(raw) != PARENT_SHA256:
        raise ValueError("Parent worker SHA256 mismatch")
    for old, new in REPLACEMENTS:
        raw = _replace(raw, old, new)
    if TARGET_SHA256 != "TO_BE_FROZEN" and sha256(raw) != TARGET_SHA256:
        raise ValueError("Target worker SHA256 mismatch")
    return raw


def restore_parent(raw):
    for old, new in reversed(REPLACEMENTS):
        raw = _replace(raw, new, old)
    if sha256(raw) != PARENT_SHA256:
        raise ValueError("Reverse patch did not recover the immutable parent")
    return raw


def main():
    raw = TARGET.read_bytes()
    patched = transform(raw)
    if restore_parent(patched) != raw:
        raise ValueError("Round-trip mismatch")
    TARGET.write_bytes(patched)
    print("parent_sha256", PARENT_SHA256)
    print("target_sha256", sha256(patched))


if __name__ == "__main__":
    main()
