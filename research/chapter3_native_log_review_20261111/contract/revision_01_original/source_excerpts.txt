    59  def _read_json_until(path, deadline, clock=time.monotonic, process=None,
    60                       sleep=time.sleep, on_poll=None):
    61      """Wait for a complete readable message without moving the deadline.
    62  
    63      Existence alone is insufficient on Windows: publication or a transient
    64      file lock can make the name visible before Python can read it.  A result
    65      is formed only after reading and decoding the full message inside the
    66      original deadline.  Diagnostics retain retries rather than hiding them.
    67      """
    68      diagnostics = {'attempts': 0, 'transient_read_errors': {}, 'last_read_error': None}
    69      while True:
    70          if clock() >= deadline:
    71              raise PublicationTimeout(path, diagnostics)
    72          if on_poll is not None:
    73              on_poll()
    74              if clock() >= deadline:
    75                  raise PublicationTimeout(path, diagnostics)
    76          diagnostics['attempts'] += 1
    77          try:
    78              value = json.loads(path.read_text(encoding='utf-8-sig'))
    79          except (FileNotFoundError, PermissionError, json.JSONDecodeError,
    80                  UnicodeDecodeError) as exc:
    81              name = type(exc).__name__
    82              diagnostics['transient_read_errors'][name] = (
    83                  diagnostics['transient_read_errors'].get(name, 0) + 1)
    84              diagnostics['last_read_error'] = f'{name}: {exc}'
    85              if isinstance(exc, FileNotFoundError) and process is not None:
    86                  returncode = process.poll()
    87                  if returncode is not None:
    88                      raise WorkerExited(returncode, diagnostics) from exc
    89              sleep(min(.005, max(0.0, deadline - clock())))
    90              continue
    91          if clock() >= deadline:
    92              raise PublicationTimeout(path, diagnostics)
    93          if not isinstance(value, dict):
    94              raise ValueError(f'MATLAB publication is not a JSON object: {path}')
    95          if on_poll is not None:
    96              on_poll()  # drain events published immediately before final response
    97              if clock() >= deadline:
    98                  raise PublicationTimeout(path, diagnostics)
    99          return value, diagnostics
   127  def _check_initial_point(payload, x0):
   128      """Check, without rounding or repairing, the optional numerical x0.
   129  
   130      These floating residuals are an admission check, not a thermal witness or
   131      independent mathematical certificate.  The receiving solver still checks
   132      x0 under its own implementation of the same declared tolerances.
   133      """
   134      info = {'provided': True, 'accepted_numerically': False,
   135              'constraint_tolerance': 1e-9, 'integer_tolerance': 1e-6,
   136              'reasons': [], 'thermal_safe_claim': False,
   137              'independent_certificate': False,
   138              'rounding_or_repair_performed': False}
   139      if np.iscomplexobj(x0):
   140          info['reasons'].append('x0 must be real; imaginary parts cannot be discarded')
   141          raise InitialPointRejected(info)
   142      try:
   143          raw = np.asarray(x0, dtype=float)
   144      except (TypeError, ValueError) as exc:
   145          info['reasons'].append('x0 cannot be converted to a numerical vector')
   146          raise InitialPointRejected(info) from exc
   147      if raw.ndim > 2 or (raw.ndim == 2 and min(raw.shape) != 1):
   148          info['reasons'].append('x0 must be a vector')
   149          raise InitialPointRejected(info)
   150      x = raw.reshape(-1)
   151      info['length'] = int(len(x))
   152      if len(x) != len(payload['c']):
   153          info['reasons'].append('x0 length differs from model variable count')
   154      if not np.isfinite(x).all():
   155          info['reasons'].append('x0 contains nonfinite values')
   156      if info['reasons']:
   157          raise InitialPointRejected(info)
   158      indices = payload['intcon'].astype(np.int64) - 1
   159      with np.errstate(over='ignore', invalid='ignore'):
   160          residual = np.asarray(payload['A'] @ x).reshape(-1) - payload['b']
   161          equality = np.asarray(payload['Aeq'] @ x).reshape(-1) - payload['beq']
   162          lower = payload['lb'] - x
   163          upper = x - payload['ub']
   164          integer = np.abs(x[indices] - np.rint(x[indices]))
   165          objective = float(payload['c'] @ x)
   166      if (not np.isfinite(residual).all() or not np.isfinite(equality).all()
   167              or not math.isfinite(objective)):
   168          info['reasons'].append('x0 produces nonfinite residuals or objective')
   169      info.update({
   170          'inequality_violation': float(max(0.0, np.max(residual, initial=0.0))),
   171          'equality_violation': float(np.max(np.abs(equality), initial=0.0)),
   172          'lower_bound_violation': float(max(0.0, np.max(lower, initial=0.0))),
   173          'upper_bound_violation': float(max(0.0, np.max(upper, initial=0.0))),
   174          'integer_violation': float(np.max(integer, initial=0.0)),
   175          'initial_objective': objective if math.isfinite(objective) else None,
   176      })
   177      for key in ('inequality_violation', 'equality_violation',
   178                  'lower_bound_violation', 'upper_bound_violation'):
   179          if info[key] > info['constraint_tolerance']:
   180              info['reasons'].append(key + ' exceeds ConstraintTolerance')
   181      if info['integer_violation'] > info['integer_tolerance']:
   182          info['reasons'].append('integer_violation exceeds IntegerTolerance')
   183      # Error diagnostics must remain valid JSON even for overflowed products.
   184      for key in ('inequality_violation', 'equality_violation',
   185                  'lower_bound_violation', 'upper_bound_violation',
   186                  'integer_violation'):
   187          if not math.isfinite(info[key]):
   188              info[key] = None
   189      if info['reasons']:
   190          raise InitialPointRejected(info)
   191      info['accepted_numerically'] = True
   192      return x, info

   195  def _prepare(model, x0=None):
   196      """Convert numerical solver input; exact legality is checked by the caller."""
   197      c = _finite_vector(model['c'], 'c')
   198      n = len(c)
   199      if n == 0:
   200          raise ValueError('Empty models must be handled before invoking solver')
   201      A = _matrix(model.get('A', sparse.csc_matrix((0, n))), 'A', n)
   202      Aeq = _matrix(model.get('Aeq', sparse.csc_matrix((0, n))), 'Aeq', n)
   203      b = _finite_vector(model.get('b', []), 'b', A.shape[0])
   204      beq = _finite_vector(model.get('beq', []), 'beq', Aeq.shape[0])
   205      lb = _finite_vector(model.get('lb', np.zeros(n)), 'lb', n, True)
   206      ub = _finite_vector(model.get('ub', np.ones(n)), 'ub', n, True)
   207      if (lb > ub).any():
   208          raise ValueError('Inconsistent variable bounds')
   209      indices = np.asarray(model.get('intcon', np.arange(n)))
   210      if indices.ndim > 1 or not np.isfinite(indices).all():
   211          raise ValueError('intcon must be a finite one-dimensional index array')
   212      if ((indices != np.floor(indices)).any() or (indices < 0).any()
   213              or (indices >= n).any() or len(set(indices.tolist())) != len(indices)):
   214          raise ValueError('intcon must contain unique zero-based integer indices')
   215      result = {'c': c, 'A': A, 'b': b, 'Aeq': Aeq, 'beq': beq,
   216                'lb': lb, 'ub': ub, 'intcon': indices.astype(float) + 1}
   217      if x0 is not None:
   218          result['x0'], result['x0_validation'] = _check_initial_point(result, x0)
   219      return result

    10      """Evidence cutoff is strict; execution stops at cooperative checkpoints.
    11

    16      def __init__(self, seconds=60, *, clock=time.monotonic, hook=None):
    17          if not math.isfinite(seconds) or seconds <= 0:
    18              raise ValueError('method seconds must be finite and positive')
    19          super().__init__(seconds=seconds, hook=hook)
    20          self.clock = clock
    21          self.start = clock()

    52              frame['exclusive_s'] = max(0.0, frame['wall_s']-frame.pop('children_s'))

    58          # receipt formed before the deadline, then checks remaining time.
